WHY THIS SHAPE (verified in repo 2026-09-13)

lib/api-zod/src/generated/api.ts is ORVAL-GENERATED from an OpenAPI spec

("Generated by orval v8.18.0 — Do not edit manually"). No generate script

exists in any package.json, and the spec file itself is not locatable from

the package metadata. Hand-editing generated/api.ts would be silently

destroyed by the next orval run.

THEREFORE: reservation request schemas go in a NEW hand-written module

(src/inventory-reservations.ts), exported alongside the generated one from

src/index.ts. generated/api.ts is NOT touched.

spec is next maintained, the 6 reservation endpoints must be added there so

orval regen does not drift from these hand-written schemas. Field names

below intentionally match the API's snake_case JSON contract so a future

spec-generated version is a drop-in replacement.

import * as zod from "zod";

export const CreateReservationBody = zod.object({
production_order_id: zod.string().uuid(),
material_id: zod.string().uuid(),
quantity: zod.coerce.number().positive(),
notes: zod.string().max(500).nullish(),
allow_partial: zod.boolean().default(false), // cap the hold at legitimately available stock
expiry_date: zod.string().datetime().nullish(),
});
export type CreateReservationBody = zod.infer;

export const AllocateReservationBody = zod.object({
quantity: zod.coerce.number().positive().optional(), // default: full unallocated remainder
lot_id: zod.string().uuid().optional(),              // force allocation from a specific lot
allow_partial: zod.boolean().default(false),         // allocate what exists instead of failing
});
export type AllocateReservationBody = zod.infer;

export const CancelReservationBody = zod.object({
reason: zod.string().min(1).max(255),
});
export type CancelReservationBody = zod.infer;

export const ReservationIdParams = zod.object({
id: zod.coerce.string().uuid(),
});
export type ReservationIdParams = zod.infer;
