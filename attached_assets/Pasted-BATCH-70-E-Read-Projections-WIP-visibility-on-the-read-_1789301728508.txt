FILE 2 — CREATE artifacts/api-server/src/routes/inventory/wip-inventory.ts

import { Router, type IRouter, type Request, type Response } from "express";
import { eq, and, asc, count, sql } from "drizzle-orm";
import {
db,
wipInventoryTable,
materialsTable,
inventoryLotsTable,
} from "@workspace/db";
import { requireAuth } from "../../middleware/auth";

const router: IRouter = Router();
// Read-only: any authenticated factory role (viewer included) — no requireWriteRole.
router.use(requireAuth);

function numify(v: unknown): unknown {
return typeof v === "string" && /^-?\d+(\.\d+)?/.test(v) ? Number(v) : v;
}

router.get("/", async (req: Request, res: Response): Promise => {
const q = req.query as Record<string, string | undefined>;
const page = Math.max(parseInt(q.page || "1", 10) || 1, 1);
const pageSize = Math.min(Math.max(parseInt(q.pageSize || "25", 10) || 25, 1), 200);
const offset = (page - 1) * pageSize;

const conds = [
q.production_order_id ? eq(wipInventoryTable.productionOrderId, q.production_order_id) : undefined,
q.material_id ? eq(wipInventoryTable.materialId, q.material_id) : undefined,
q.status ? eq(wipInventoryTable.status, q.status) : undefined,
].filter(Boolean) as any[];

const where = conds.length ? and(...conds) : undefined;
const [t] = await db.select({ count: count() }).from(wipInventoryTable).where(where);
const rows = await db
.select({
wip: wipInventoryTable,
materialCode: materialsTable.code,
materialName: materialsTable.name,
lotNumber: inventoryLotsTable.lotNumber,
})
.from(wipInventoryTable)
.innerJoin(materialsTable, eq(materialsTable.id, wipInventoryTable.materialId))
.innerJoin(inventoryLotsTable, eq(inventoryLotsTable.id, wipInventoryTable.lotId))
.where(where)
.orderBy(asc(wipInventoryTable.createdAt))
.limit(pageSize)
.offset(offset);

const total = Number(t?.count ?? 0);
res.json({
items: rows.map(({ wip: w, materialCode, materialName, lotNumber }) => ({
id: w.id,
production_order_id: w.productionOrderId,
material_id: w.materialId,
material_code: materialCode,
material_name: materialName,
lot_id: w.lotId,
lot_number: lotNumber,
warehouse_id: w.warehouseId,
location_id: w.locationId,
issued_qty: numify(w.issuedQty),
consumed_qty: numify(w.consumedQty),
returned_qty: numify(w.returnedQty),
scrapped_qty: numify(w.scrappedQty),
remaining_qty: numify(w.remainingQty),
uom: w.uom,
status: w.status,
wip_issue_note_id: w.wipIssueNoteId ?? null,
issue_id: w.issueId ?? null,
created_at: w.createdAt,
})),
meta: { total, page, pageSize, totalPages: Math.ceil(total / pageSize) },
});
});

export default router;
