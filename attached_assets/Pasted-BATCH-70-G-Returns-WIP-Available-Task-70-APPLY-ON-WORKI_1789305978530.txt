PHYSICAL MODEL (D6)

POST (the only physical op): −qty ledger 'wip' +qty 'available', type RETURN,

per wip-row with that row's lot dims; wip row returnedQty += take, remaining

recalculated (existing CHECK), status fully_consumed at remaining 0 (shared

terminal with consumption). Destination dims on the +available row from the

document (stock returns to the SAME lot/GRN-line pool, new location).

Flow: draft → approved → posted | rejected. Approval is a real gate.

Idempotency: Idempotency-Key header on POST; in-tx advisory lock

('return-idem:') + unique column + dual-shape 23505 replay (the 70-B

TOCTOU-hardened pattern — pre-tx SELECT is FORBIDDEN).

Locks: document FOR UPDATE → wip rows FOR UPDATE (createdAt order — the same

surface consumption locks, so return↔consumption races serialize; loser 409,

ZERO mutation).

function numify(v: unknown): unknown {
return typeof v === "string" && /^-?\d+(\.\d+)?/.test(v) ? Number(v) : v;
}
function todayDateStr(): string {
const n = new Date();

id: r.id, return_number: r.returnNumber,
production_order_id: r.productionOrderId, material_id: r.materialId,

router.post("/", requireWriteRole("supervisor", "director"), async (req: Request, res: Response): Promise => {
const parsed = CreateReturnBody.safeParse(req.body);
if (!parsed.success) { res.status(400).json({ error: parsed.error.message }); return; }
const b = parsed.data;
const actorId = req.user?.userId ?? null;

const [po] = await db.select({ id: mfgProductionOrdersTable.id }).from(mfgProductionOrdersTable)
.where(eq(mfgProductionOrdersTable.id, b.production_order_id)).limit(1);
if (!po) { res.status(400).json({ error: "Unknown production order id" }); return; }
const [mat] = await db.select({ id: materialsTable.id, uom: materialsTable.uom }).from(materialsTable)
.where(eq(materialsTable.id, b.material_id)).limit(1);
if (!mat) { res.status(400).json({ error: "Unknown material id" }); return; }
const [wh] = await db.select({ id: warehousesTable.id }).from(warehousesTable)
.where(eq(warehousesTable.id, b.destination_warehouse_id)).limit(1);
if (!wh) { res.status(400).json({ error: "Unknown destination warehouse id" }); return; }
const [note] = await db.select().from(wipIssueNotesTable)
.where(eq(wipIssueNotesTable.id, b.wip_issue_note_id)).limit(1);

res.status(422).json({ error: "WIP issue note does not belong to this production order" }); return;
}
if (b.lot_id) {
const ln = await db.select({ lotId: wipIssueLinesTable.lotId }).from(wipIssueLinesTable)

if (!ln.some((l) => l.lotId === b.lot_id)) { res.status(422).json({ error: "Lot is not part of this issue note" }); return; }

// Source warehouse derived from the note's wip rows (never client-supplied).
const wipRows = await db.select({ wh: wipInventoryTable.warehouseId }).from(wipInventoryTable)
.where(eq(wipInventoryTable.wipIssueNoteId, note.id));
const sources = [...new Set(wipRows.map((w) => w.wh))];

productionOrderId: b.production_order_id,
materialId: b.material_id,
lotId: b.lot_id ?? null,

// ─── Post (PHYSICAL: −wip / +available) ─────────────────────────────────────
router.post("/:id/post", requireWriteRole("supervisor", "director"), async (req: Request, res: Response): Promise => {
const actorId = req.user?.userId ?? null;
const idemKey = (req.header("idempotency-key") ?? "").slice(0, 100) || null;

const outcome = await db.transaction(async (tx) => {
// TOCTOU-hardened idempotency (70-B pattern): lock first, select under lock,
// insert, dual-shape 23505 fallback. No pre-tx lookup.
if (idemKey) {
await tx.execute(sql`SELECT pg_advisory_xact_lock(hashtext(${`return-idem:{idemKey}`}))`);
const [replay] = await tx.select().from(returnDocumentsTable)
.where(eq(returnDocumentsTable.idempotencyKey, idemKey)).limit(1);
if (replay) return { status: "replay" as const, doc: replay };
}

    const [r] = await tx.select().from(returnDocumentsTable)
      .where(eq(returnDocumentsTable.id, req.params.id as string)).for("update").limit(1);
    if (!r) return { status: "not_found" as const };
    if (r.status === "posted") return { status: "already" as const };
    if (r.status !== "approved") return { status: "invalid_state" as const, current: r.status };

    const qty = Number(r.quantity);
    const wipConds = [
      eq(wipInventoryTable.wipIssueNoteId, r.wipIssueNoteId),
      sql`${wipInventoryTable.status} IN ('active','partially_consumed')`,
      sql`${wipInventoryTable.remainingQty}::numeric > 0`,
    ];
    if (r.lotId) wipConds.push(eq(wipInventoryTable.lotId, r.lotId));
    const wipRows = await tx.select().from(wipInventoryTable)
      .where(and(...wipConds)).orderBy(asc(wipInventoryTable.createdAt)).for("update");

    const totalRemaining = wipRows.reduce((s, w) => s + Number(w.remainingQty), 0);
    if (qty > totalRemaining) return { status: "over_return" as const, totalRemaining };

    let remaining = qty;
    for (const w of wipRows) {
      if (remaining <= 0) break;
      const take = Math.min(remaining, Number(w.remainingQty));
      const returned = Number(w.returnedQty) + take;
      const left = Number(w.issuedQty) - Number(w.consumedQty) - returned - Number(w.scrappedQty);
      const [lot] = await tx.select().from(inventoryLotsTable)
        .where(eq(inventoryLotsTable.id, w.lotId)).limit(1);
      await tx.update(wipInventoryTable).set({
        returnedQty: String(returned),
        remainingQty: String(left),             // existing CHECK satisfied exactly
        status: left <= 0 ? "fully_consumed" : "partially_consumed",
      }).where(eq(wipInventoryTable.id, w.id));
      // D6: −wip (source dims) / +available (destination dims), same lot/line pool.
      await tx.insert(inventoryTransactionsTable).values({
        materialId: r.materialId, quantity: String(-take), uom: r.uom,
        stockState: "wip", transactionType: "RETURN",
        sourceDocumentType: "return_document", sourceDocumentId: r.id,
        sourceLineId: lot?.grnLineId ?? null,
        lotId: w.lotId, warehouseId: w.warehouseId, locationId: w.locationId,
        productionOrderId: r.productionOrderId, actorId, actorName: req.user?.email ?? null,
      });
      await tx.insert(inventoryTransactionsTable).values({
        materialId: r.materialId, quantity: String(take), uom: r.uom,
        stockState: "available", transactionType: "RETURN",
        sourceDocumentType: "return_document", sourceDocumentId: r.id,
        sourceLineId: lot?.grnLineId ?? null,
        lotId: w.lotId, warehouseId: r.destinationWarehouseId,
        locationId: r.destinationLocationId, binId: r.destinationBinId,
        productionOrderId: r.productionOrderId, actorId, actorName: req.user?.email ?? null,
      });
      remaining -= take;
    }

    const [updated] = await tx.update(returnDocumentsTable).set({
      status: "posted", postedBy: actorId, postedAt: new Date(),
      idempotencyKey: idemKey,
    }).where(eq(returnDocumentsTable.id, r.id)).returning();

    await tx.insert(outboxEventsTable).values({
      aggregateType: "return_document", aggregateId: r.id,
      eventType: "RETURN_POSTED",
      payload: {
        return_number: r.returnNumber, production_order_id: r.productionOrderId,
        material_id: r.materialId, wip_issue_note_id: r.wipIssueNoteId,
        quantity: qty, actor_id: actorId,
      },
    });
    return { status: "ok" as const, doc: updated };

});

if (outcome.status === "not_found") { res.status(404).json({ error: "Return not found" }); return; }
if (outcome.status === "already") { res.status(409).json({ error: "Return is already posted" }); return; }
if (outcome.status === "replay") { res.status(200).json(serializeReturn(outcome.doc as Record<string, any>)); return; }
if (outcome.status === "invalid_state") { res.status(409).json({ error: `Return cannot be posted from status '${outcome.current}'` }); return; }
if (outcome.status === "over_return") {
res.status(409).json({ error: "OVER_RETURN", details: { requested: undefined, total_remaining: outcome.totalRemaining } }); return;
}

void recordSecurityEvent({
eventType: "return.posted", actorId, actorEmail: req.user?.email ?? null,
actorRole: req.user?.role ?? null, ...reqMeta(req), statusCode: 200,
detail: `Return ${outcome.doc.returnNumber} posted (−wip/+available)`,
});
res.json(serializeReturn(outcome.doc as Record<string, any>));
});
