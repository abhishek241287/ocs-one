BATCH 70-C — Consumption (draft → confirm, −wip accounting, variance)

Task #70 · APPLY ON WORKING TREE · dev DB only · NO COMMIT.

Depends on: 70-A v2 (schema), 70-B (issue engine — certified pending your go).

Ratified model: D2 consumption = −wip ledger rows; returns/scrap movements stay

OUT of scope (their table columns remain 0 — accepting them without the matching

stock movement would corrupt the wip CHECK, so the API rejects them for now).

---------------------------------------------------------------------------

FLOW

POST /api/inventory/consumptions            → draft confirmation

(planned_qty from BOM snapshot line when available, else body override,

else actual; variance_qty = actual − planned enforced by DB CHECK)

POST /api/inventory/consumptions/:id/confirm → the ONLY physical mutation:

locks confirmation + wip rows FOR UPDATE (deterministic createdAt order),

validates actual ≤ remaining wip, writes −qty stock_state='wip' ledger

rows (transactionType CONSUMPTION, per wip-row with that row's lot dims),

updates wip_inventory (consumedQty/remainingQty/status), sets

status confirmed + confirmed_by/at, outbox CONSUMPTION_CONFIRMED.

GET  /api/inventory/consumptions[/:id]       → reads for verification.

Re-confirm → 409 (deterministic, like release). No idempotency key in 70-C:

the row lock + draft→confirmed transition gives at-most-once (double-submit

→ one 200, one 409; retried 409 surfaces the completed state, no re-write).

CONCURRENCY: confirm takes confirmation FOR UPDATE first, then wip rows FOR

UPDATE in createdAt order. Issue (70-B) locks wip rows only via the reservation

chain (reservation → allocations → line), NOT wip rows directly... therefore a

concurrent issue+consumption on the same wip row is ordered by the wip row

lock itself whichever arrives first (both take FOR UPDATE on the same rows —

issue inserts NEW wip rows per note, consumption updates EXISTING rows; the

only shared surface is reading remaining, which each recomputes under lock).

Documented for 70-F.

import { Router, type IRouter, type Request, type Response } from "express";
import { eq, and, asc, count, sql } from "drizzle-orm";
import {
db,
pool,
consumptionConfirmationsTable,
wipInventoryTable,
inventoryTransactionsTable,
materialsTable,
mfgProductionOrdersTable,
bomSnapshotsTable,
bomSnapshotLinesTable,
outboxEventsTable,
} from "@workspace/db";
import { CreateConsumptionBody, ConfirmConsumptionBody } from "@workspace/api-zod";
import { requireWriteRole } from "../../middleware/auth";
import { recordSecurityEvent, reqMeta } from "../../lib/security-events";

const router: IRouter = Router();
router.use(requireWriteRole("supervisor", "director")); // reads pass for any authed user

function numify(v: unknown): unknown {
return typeof v === "string" && /^-?\d+(\.\d+)?/.test(v) ? Number(v) : v;
}

function todayDateStr(): string {
const n = new Date();
return `${n.getUTCFullYear()}${String(n.getUTCMonth() + 1).padStart(2, "0")}${String(n.getUTCDate()).padStart(2, "0")}`;
}

function serializeConfirmation(c: Record<string, any>) {
return {
id: c.id,
confirmation_number: c.confirmationNumber,
production_order_id: c.productionOrderId,
material_id: c.materialId,
lot_id: c.lotId ?? null,
planned_qty: numify(c.plannedQty),
actual_qty: numify(c.actualQty),
variance_qty: numify(c.varianceQty),
status: c.status,
reason: c.reason ?? null,
confirmed_by: c.confirmedBy ?? null,
confirmed_at: c.confirmedAt ?? null,
created_at: c.createdAt,
};
}

// ─── Create draft ────────────────────────────────────────────────────────────
router.post("/", async (req: Request, res: Response): Promise => {
const parsed = CreateConsumptionBody.safeParse(req.body);
if (!parsed.success) { res.status(400).json({ error: parsed.error.message }); return; }
const body = parsed.data;
const actorId = req.user?.userId ?? null;

const [po] = await db.select({ id: mfgProductionOrdersTable.id })
.from(mfgProductionOrdersTable).where(eq(mfgProductionOrdersTable.id, body.production_order_id)).limit(1);
if (!po) { res.status(400).json({ error: "Unknown production order id" }); return; }
const [mat] = await db.select({ id: materialsTable.id })
.from(materialsTable).where(eq(materialsTable.id, body.material_id)).limit(1);
if (!mat) { res.status(400).json({ error: "Unknown material id" }); return; }

// Planned qty: explicit override > BOM snapshot line > actual (variance 0).
let planned = body.planned_qty ?? null;
if (planned === null) {
const [snap] = await db.select({ id: bomSnapshotsTable.id })
.from(bomSnapshotsTable)
.where(eq(bomSnapshotsTable.productionOrderId, body.production_order_id))
.limit(1);
if (snap) {
const [line] = await db.select({ quantity: bomSnapshotLinesTable.quantity })
.from(bomSnapshotLinesTable)
.where(and(eq(bomSnapshotLinesTable.bomSnapshotId, snap.id),
eq(bomSnapshotLinesTable.materialId, body.material_id)))
.limit(1);
planned = line ? Number(line.quantity) : null;
}
}
if (planned === null) planned = body.actual_qty;

const { rows } = await pool.query("SELECT nextval('consumption_seq') AS seq");
const confirmationNumber = `CC-${todayDateStr()}-${String(rows[0].seq).padStart(4, "0")}`;

const [created] = await db.insert(consumptionConfirmationsTable).values({
confirmationNumber,
productionOrderId: body.production_order_id,
materialId: body.material_id,
lotId: body.lot_id ?? null,
plannedQty: String(planned),
actualQty: String(body.actual_qty),
varianceQty: String(body.actual_qty - planned), // DB CHECK enforces the identity
reason: body.reason ?? null,
status: "draft",
createdBy: actorId, // ADJUST: column name per inventory-platform.ts
}).returning();

void recordSecurityEvent({
eventType: "consumption.draft_created", actorId,
actorEmail: req.user?.email ?? null, actorRole: req.user?.role ?? null,
...reqMeta(req), statusCode: 201,
detail: `Consumption draft ${confirmationNumber} (PO ${body.production_order_id}, actual ${body.actual_qty}, planned ${planned})`,
});

res.status(201).json(serializeConfirmation(created as Record<string, any>));
});

// ─── Confirm (the physical mutation) ────────────────────────────────────────
router.post("/:id/confirm", async (req: Request, res: Response): Promise => {
const parsed = ConfirmConsumptionBody.safeParse(req.body ?? {});
if (!parsed.success) { res.status(400).json({ error: parsed.error.message }); return; }
const actorId = req.user?.userId ?? null;

const outcome = await db.transaction(async (tx) => {
const [c] = await tx.select().from(consumptionConfirmationsTable)
.where(eq(consumptionConfirmationsTable.id, req.params.id as string))
.for("update").limit(1);
if (!c) return { status: "not_found" as const };
if (c.status !== "draft") return { status: "invalid_state" as const, current: c.status };

    const actual = Number(c.actualQty);
    const wipConds = [
      eq(wipInventoryTable.productionOrderId, c.productionOrderId),
      eq(wipInventoryTable.materialId, c.materialId),
      sql`${wipInventoryTable.status} IN ('active','partially_consumed')`,
      sql`${wipInventoryTable.remainingQty}::numeric > 0`,
    ];
    if (c.lotId) wipConds.push(eq(wipInventoryTable.lotId, c.lotId));
    const wipRows = await tx.select().from(wipInventoryTable)
      .where(and(...wipConds))
      .orderBy(asc(wipInventoryTable.createdAt))
      .for("update");

    const totalRemaining = wipRows.reduce((s, w) => s + Number(w.remainingQty), 0);
    if (actual > totalRemaining) return { status: "insufficient_wip" as const, totalRemaining };

    let remaining = actual;
    for (const w of wipRows) {
      if (remaining <= 0) break;
      const take = Math.min(remaining, Number(w.remainingQty));
      const consumed = Number(w.consumedQty) + take;
      const left = Number(w.issuedQty) - consumed - Number(w.returnedQty) - Number(w.scrappedQty);
      await tx.update(wipInventoryTable).set({
        consumedQty: String(consumed),
        remainingQty: String(left), // satisfies the CHECK exactly
        status: left <= 0 ? "fully_consumed" : "partially_consumed",
      }).where(eq(wipInventoryTable.id, w.id));
      await tx.insert(inventoryTransactionsTable).values({
        materialId: c.materialId,
        quantity: String(-take),
        uom: w.uom,
        stockState: "wip",
        transactionType: "CONSUMPTION",
        sourceDocumentType: "consumption_confirmation",
        sourceDocumentId: c.id,
        lotId: w.lotId,
        warehouseId: w.warehouseId,
        locationId: w.locationId,
        productionOrderId: c.productionOrderId,
        actorId,
        actorName: req.user?.email ?? null,
      });
      remaining -= take;
    }

    const [updated] = await tx.update(consumptionConfirmationsTable).set({
      status: "confirmed",
      confirmedBy: actorId,
      confirmedAt: new Date(),
    }).where(eq(consumptionConfirmationsTable.id, c.id)).returning();

    await tx.insert(outboxEventsTable).values({
      aggregateType: "consumption_confirmation",
      aggregateId: c.id,
      eventType: "CONSUMPTION_CONFIRMED",
      payload: {
        confirmation_number: c.confirmationNumber,
        production_order_id: c.productionOrderId,
        material_id: c.materialId,
        lot_id: c.lotId,
        actual_qty: actual,
        planned_qty: Number(c.plannedQty),
        variance_qty: Number(c.varianceQty),
        actor_id: actorId,
      },
    });

    return { status: "ok" as const, updated };

});

if (outcome.status === "not_found") { res.status(404).json({ error: "Consumption confirmation not found" }); return; }
if (outcome.status === "invalid_state") { res.status(409).json({ error: `Confirmation cannot be confirmed from status '${outcome.current}'` }); return; }
if (outcome.status === "insufficient_wip") {
res.status(409).json({ error: "INSUFFICIENT_WIP", details: { actual: undefined, total_remaining: outcome.totalRemaining } });
return;
}

void recordSecurityEvent({
eventType: "consumption.confirmed", actorId,
actorEmail: req.user?.email ?? null, actorRole: req.user?.role ?? null,
...reqMeta(req), statusCode: 200,
detail: `Consumption ${outcome.updated.confirmationNumber} confirmed`,
});

res.json(serializeConfirmation(outcome.updated as Record<string, any>));
});

// ─── Reads ───────────────────────────────────────────────────────────────────
router.get("/", async (req: Request, res: Response): Promise => {
const q = req.query as Record<string, string | undefined>;
const conds = [
q.production_order_id ? eq(consumptionConfirmationsTable.productionOrderId, q.production_order_id) : undefined,
q.material_id ? eq(consumptionConfirmationsTable.materialId, q.material_id) : undefined,
q.status ? eq(consumptionConfirmationsTable.status, q.status) : undefined,
].filter(Boolean) as any[];
const where = conds.length ? and(...conds) : undefined;
const [t] = await db.select({ count: count() }).from(consumptionConfirmationsTable).where(where);
const rows = await db.select().from(consumptionConfirmationsTable).where(where)
.orderBy(asc(consumptionConfirmationsTable.createdAt)).limit(200);
res.json({ items: rows.map((r) => serializeConfirmation(r as Record<string, any>)), meta: { total: Number(t?.count ?? 0) } });
});

router.get("/:id", async (req: Request, res: Response): Promise => {
const [c] = await db.select().from(consumptionConfirmationsTable)
.where(eq(consumptionConfirmationsTable.id, req.params.id as string)).limit(1);
if (!c) { res.status(404).json({ error: "Consumption confirmation not found" }); return; }
res.json(serializeConfirmation(c as Record<string, any>));
});

export default router;
