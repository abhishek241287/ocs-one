-- Replace the legacy unique (exact confirmed name) with a COALESCE identity unique
-- covering BOTH legacy MIN links (issue_id) and Phase 4 links (wip_issue_note_id).
-- MIN never writes wip_inventory (audit-verified), so this is zero-behavior-change;
-- the table is currently empty.
DROP INDEX IF EXISTS wip_inventory_order_material_lot_issue_unique;
CREATE UNIQUE INDEX wip_inventory_identity_unique
ON wip_inventory (
production_order_id,
material_id,
COALESCE(lot_id,            '00000000-0000-0000-0000-000000000000'::uuid),
COALESCE(wip_issue_note_id, issue_id, '00000000-0000-0000-0000-000000000000'::uuid)
);

-- WIP / consumption reconciliation read paths.
CREATE INDEX IF NOT EXISTS idx_wip_inventory_po_status   ON wip_inventory (production_order_id, status);
CREATE INDEX IF NOT EXISTS idx_wip_inventory_mat_status  ON wip_inventory (material_id, status);
CREATE INDEX IF NOT EXISTS idx_consumption_po_status     ON consumption_confirmations (production_order_id, status);
CREATE INDEX IF NOT EXISTS idx_it_po_state               ON inventory_transactions (production_order_id, stock_state);
