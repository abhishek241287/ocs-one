the wip rows (never client-supplied), same as 70-G.

(PO, material[, lot])'s wip rows — same granularity as consumption (70-C).

Races: scrap↔consumption, scrap↔return (70-G) serialize on wip rows;

function numify(v: unknown): unknown {
return typeof v === "string" && /^-?\d+(\.\d+)?/.test(v) ? Number(v) : v;
}
function todayDateStr(): string {
const n = new Date();
return `${n.getUTCFullYear()}${String(n.getUTCMonth() + 1).padStart(2, "0")}${String(n.getUTCDate()).padStart(2, "0")}`;
}
function serializeScrap(r: Record<string, any>) {
return {
id: r.id, scrap_number: r.scrapNumber,
production_order_id: r.productionOrderId, material_id: r.materialId,
lot_id: r.lotId ?? null, stage_id: r.stageId ?? null,
serial_number: r.serialNumber ?? null,
source_warehouse_id: r.sourceWarehouseId,
source_location_id: r.sourceLocationId ?? null,
quantity: numify(r.quantity), uom: r.uom,
reason: r.reason, notes: r.notes ?? null, status: r.status,
approved_by: r.approvedBy ?? null, approved_at: r.approvedAt ?? null,
posted_by: r.postedBy ?? null, posted_at: r.postedAt ?? null,
created_by: r.createdBy, created_at: r.createdAt,

}

router.post("/", requireWriteRole("supervisor", "director"), async (req: Request, res: Response): Promise => {
const parsed = CreateScrapBody.safeParse(req.body);
if (!parsed.success) { res.status(400).json({ error: parsed.error.message }); return; }
const b = parsed.data;
const actorId = req.user?.userId ?? null;

const [po] = await db.select({ id: mfgProductionOrdersTable.id }).from(mfgProductionOrdersTable)
.where(eq(mfgProductionOrdersTable.id, b.production_order_id)).limit(1);
if (!po) { res.status(400).json({ error: "Unknown production order id" }); return; }
const [mat] = await db.select({ id: materialsTable.id, uom: materialsTable.uom }).from(materialsTable)
.where(eq(materialsTable.id, b.material_id)).limit(1);
if (!mat) { res.status(400).json({ error: "Unknown material id" }); return; }

// Source warehouse/location derived from the (PO, material[, lot]) wip rows.
const wipConds = [
eq(wipInventoryTable.productionOrderId, b.production_order_id),
eq(wipInventoryTable.materialId, b.material_id),
];
if (b.lot_id) wipConds.push(eq(wipInventoryTable.lotId, b.lot_id));
const wipRows = await db.select({
wh: wipInventoryTable.warehouseId, loc: wipInventoryTable.locationId,
}).from(wipInventoryTable).where(and(...wipConds));
if (!wipRows.length) { res.status(422).json({ error: "No WIP found for this production order/material/lot" }); return; }
const whs = [...new Set(wipRows.map((w) => w.wh))];
if (whs.length !== 1) { res.status(422).json({ error: "WIP spans multiple source warehouses; narrow by lot" }); return; }
const locs = [...new Set(wipRows.map((w) => w.loc))];
const sourceLoc = locs.length === 1 ? locs[0] : null;

const { rows } = await pool.query("SELECT nextval('scrap_seq') AS seq");
const scrapNumber = `SCR-${todayDateStr()}-${String(rows[0].seq).padStart(4, "0")}`;

const [created] = await db.insert(scrapDocumentsTable).values({

// ─── Post (PHYSICAL: −wip / +scrapped) ──────────────────────────────────────
router.post("/:id/post", requireWriteRole("supervisor", "director"), async (req: Request, res: Response): Promise => {
const actorId = req.user?.userId ?? null;
const idemKey = (req.header("idempotency-key") ?? "").slice(0, 100) || null;

const outcome = await db.transaction(async (tx) => {
if (idemKey) {
await tx.execute(sql`SELECT pg_advisory_xact_lock(hashtext(${`scrap-idem:{idemKey}`}))`);
const [replay] = await tx.select().from(scrapDocumentsTable)
.where(eq(scrapDocumentsTable.idempotencyKey, idemKey)).limit(1);
if (replay) return { status: "replay" as const, doc: replay };
}
const [r] = await tx.select().from(scrapDocumentsTable)
.where(eq(scrapDocumentsTable.id, req.params.id as string)).for("update").limit(1);
if (!r) return { status: "not_found" as const };
if (r.status === "posted") return { status: "already" as const };
if (r.status !== "approved") return { status: "invalid_state" as const, current: r.status };

    const qty = Number(r.quantity);
    const wipConds = [
      eq(wipInventoryTable.productionOrderId, r.productionOrderId),
      eq(wipInventoryTable.materialId, r.materialId),
      sql`${wipInventoryTable.status} IN ('active','partially_consumed')`,
      sql`${wipInventoryTable.remainingQty}::numeric > 0`,
    ];
    if (r.lotId) wipConds.push(eq(wipInventoryTable.lotId, r.lotId));
    const wipRows = await tx.select().from(wipInventoryTable)
      .where(and(...wipConds)).orderBy(asc(wipInventoryTable.createdAt)).for("update");

    const totalRemaining = wipRows.reduce((s, w) => s + Number(w.remainingQty), 0);
    if (qty > totalRemaining) return { status: "over_scrap" as const, totalRemaining };

    let remaining = qty;
    for (const w of wipRows) {
      if (remaining <= 0) break;
      const take = Math.min(remaining, Number(w.remainingQty));
      const scrapped = Number(w.scrappedQty) + take;
      const left = Number(w.issuedQty) - Number(w.consumedQty) - Number(w.returnedQty) - scrapped;
      const [lot] = await tx.select().from(inventoryLotsTable)
        .where(eq(inventoryLotsTable.id, w.lotId)).limit(1);
      await tx.update(wipInventoryTable).set({
        scrappedQty: String(scrapped),
        remainingQty: String(left),
        status: left <= 0 ? "fully_consumed" : "partially_consumed",
      }).where(eq(wipInventoryTable.id, w.id));
      // D7: −wip (source dims) / +scrapped (same dims). Zero-value visible state.
      await tx.insert(inventoryTransactionsTable).values({
        materialId: r.materialId, quantity: String(-take), uom: r.uom,
        stockState: "wip", transactionType: "SCRAP",
        sourceDocumentType: "scrap_document", sourceDocumentId: r.id,
        sourceLineId: lot?.grnLineId ?? null,
        lotId: w.lotId, warehouseId: w.warehouseId, locationId: w.locationId,
        productionOrderId: r.productionOrderId, actorId, actorName: req.user?.email ?? null,
      });
      await tx.insert(inventoryTransactionsTable).values({
        materialId: r.materialId, quantity: String(take), uom: r.uom,
        stockState: "scrapped", transactionType: "SCRAP",
        sourceDocumentType: "scrap_document", sourceDocumentId: r.id,
        sourceLineId: lot?.grnLineId ?? null,
        lotId: w.lotId, warehouseId: w.warehouseId, locationId: w.locationId,
        productionOrderId: r.productionOrderId, actorId, actorName: req.user?.email ?? null,
      });
      remaining -= take;
    }

    const [updated] = await tx.update(scrapDocumentsTable).set({
      status: "posted", postedBy: actorId, postedAt: new Date(),
      idempotencyKey: idemKey,
    }).where(eq(scrapDocumentsTable.id, r.id)).returning();

    await tx.insert(outboxEventsTable).values({
      aggregateType: "scrap_document", aggregateId: r.id,
      eventType: "SCRAP_POSTED",
      payload: {
        scrap_number: r.scrapNumber, production_order_id: r.productionOrderId,
        material_id: r.materialId, lot_id: r.lotId, quantity: qty, actor_id: actorId,
      },
    });
    return { status: "ok" as const, doc: updated };

});

if (outcome.status === "not_found") { res.status(404).json({ error: "Scrap not found" }); return; }
if (outcome.status === "already") { res.status(409).json({ error: "Scrap is already posted" }); return; }
if (outcome.status === "replay") { res.status(200).json(serializeScrap(outcome.doc as Record<string, any>)); return; }
if (outcome.status === "invalid_state") { res.status(409).json({ error: `Scrap cannot be posted from status '${outcome.current}'` }); return; }
if (outcome.status === "over_scrap") {
res.status(409).json({ error: "OVER_SCRAP", details: { requested: undefined, total_remaining: outcome.totalRemaining } }); return;
}

void recordSecurityEvent({
eventType: "scrap.posted", actorId, actorEmail: req.user?.email ?? null,
actorRole: req.user?.role ?? null, ...reqMeta(req), statusCode: 200,
detail: `Scrap ${outcome.doc.scrapNumber} posted (−wip/+scrapped)`,
});
res.json(serializeScrap(outcome.doc as Record<string, any>));
});
