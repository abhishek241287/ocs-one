MOVEMENT-MODEL FACTS THIS BATCH IS BUILT ON (verified in code 2026-09-13)

- Put-away writes NO ledger rows (grns.ts put-away comment: "the accepted

quantity is already available stock"). Ledger available for a receipt lives

at LINE level: SUM(quantity) WHERE source_line_id=? AND stock_state='available'.

- Option A transfers decrement that SAME line-level balance with a −qty ledger

row and lock the GRN line FOR UPDATE. → allocation MUST lock the GRN line

FOR UPDATE (same object) to serialize against transfers. Locking only the

lot row would NOT block a concurrent transfer (TOCTOU).

- grn_line_items.lot_id is the line's internal lot (put-away requires it);

lots also have their own status (active/quarantined/rejected/consumed/expired)

changed WITHOUT ledger writes (lots.ts actions) → allocation must filter

lot.status='active' AND use ledger-line availability (RES-12).

- inventory_lots.expiry_date EXISTS (populated when source data provides it) →

FEFO is now honestly supportable: ORDER BY expiry_date ASC NULLS LAST,

received_date ASC. Default strategy stays FIFO (RES-04); strategy: "FEFO"

is opt-in. Never fake FEFO — lots without expiry simply sort last.

- 1 line : N lots possible (lot split) → per-lot reservable is derived from the

LINE balance minus ALL active allocations on that line's lots, attributed to

lots in deterministic scan order (never allocates more than the line holds).

// ─── allocation helpers ──────────────────────────────────────────────────────
// House availability formula: the signed ledger at LINE level (the same formula
// transfers.ts and the cell-stock picker use). Never lots.remaining_qty.
async function lineAvailable(tx: any, grnLineId: string): Promise {
const [row] = await tx
.select({ sum: sql`coalesce(sum(${inventoryTransactionsTable.quantity}), 0)` })
.from(inventoryTransactionsTable)
.where(
and(
eq(inventoryTransactionsTable.sourceLineId, grnLineId),
eq(inventoryTransactionsTable.stockState, "available"),
),
);
return Number(row.sum);
}

// Active allocation holds on ALL lots of a line (lots may be split N:1 from a line).
async function lineHeld(tx: any, grnLineId: string): Promise {
const [row] = await tx
.select({
sum: sql`coalesce(sum(${inventoryReservationAllocationsTable.quantity}), 0)`,
})
.from(inventoryReservationAllocationsTable)
.innerJoin(
inventoryLotsTable,
eq(inventoryLotsTable.id, inventoryReservationAllocationsTable.lotId),
)
.where(
and(
eq(inventoryLotsTable.grnLineId, grnLineId),
eq(inventoryReservationAllocationsTable.status, "active"),
),
);
return Number(row.sum);
}

function nextStatus(reserved: number, allocated: number): string {
if (allocated >= reserved) return "fully_allocated";
if (allocated > 0) return "partially_allocated";
return "active";
}

// ─── Allocate ────────────────────────────────────────────────────────────────
router.post("/:id/allocate", async (req: Request, res: Response): Promise => {
const parsed = AllocateReservationBody.safeParse(req.body ?? {});
if (!parsed.success) {
res.status(400).json({ error: parsed.error.message });
return;
}
const body = parsed.data;
const actorId = req.user?.userId ?? null;

const outcome = await db.transaction(async (tx) => {
const [reservation] = await tx
.select()
.from(inventoryReservationsTable)
.where(eq(inventoryReservationsTable.id, req.params.id as string))
.for("update")
.limit(1);
if (!reservation) return { status: "not_found" as const };
// Lazy expiry (no background job in this codebase).
if (
reservation.expiresAt &&
new Date(reservation.expiresAt) < new Date() &&
["active", "partially_allocated", "fully_allocated"].includes(reservation.status)
) {
await tx
.update(inventoryReservationsTable)
.set({ status: "expired" })
.where(eq(inventoryReservationsTable.id, reservation.id));
return { status: "expired" as const };
}
if (!["active", "partially_allocated"].includes(reservation.status)) {
return { status: "invalid_state" as const, current: reservation.status };
}

    const reserved = Number(reservation.reservedQty);
    const allocated = Number(reservation.allocatedQty);
    const need0 = body.quantity ?? reserved - allocated;
    if (need0 <= 0) return { status: "nothing" as const };
    let need = need0;

    // Optional forced lot (must belong to the reserved material).
    if (body.lot_id) {
      const [forced] = await tx
        .select({ id: inventoryLotsTable.id, materialId: inventoryLotsTable.materialId })
        .from(inventoryLotsTable)
        .where(eq(inventoryLotsTable.id, body.lot_id))
        .limit(1);
      if (!forced || forced.materialId !== reservation.materialId) {
        return { status: "bad_lot" as const };
      }
    }

    // Candidate lots: same material, ACTIVE lot status only (quarantined/rejected/
    // consumed/expired are never allocatable — RES-12). Optional forced-lot filter.
    // Lock order is the scan order below → deterministic → deadlock-free between
    // concurrent allocators; transfers take the same per-line FOR UPDATE lock.
    const lotConds = [
      eq(inventoryLotsTable.materialId, reservation.materialId),
      eq(inventoryLotsTable.status, "active"),
    ];
    if (body.lot_id) lotConds.push(eq(inventoryLotsTable.id, body.lot_id));

    const orderBy =
      body.strategy === "FEFO"
        ? [asc(sql`${inventoryLotsTable.expiryDate} ASC NULLS LAST`),
           asc(inventoryLotsTable.receivedDate),
           asc(inventoryLotsTable.lotNumber)]
        : [asc(inventoryLotsTable.receivedDate),
           asc(inventoryLotsTable.lotNumber)];

    const candidates = await tx
      .select({
        id: inventoryLotsTable.id,
        grnLineId: inventoryLotsTable.grnLineId,
        lotNumber: inventoryLotsTable.lotNumber,
        warehouseId: inventoryLotsTable.warehouseId,
        locationId: inventoryLotsTable.locationId,
        binId: inventoryLotsTable.binId,
      })
      .from(inventoryLotsTable)
      .where(and(...lotConds))
      .orderBy(...orderBy)
      .for("update");

    const allocations: Array<{
      lotId: string; grnLineId: string; lotNumber: string;
      warehouseId: string | null; locationId: string | null; binId: string | null;
      quantity: number;
    }> = [];
    const lockedLines = new Map<string, number>(); // grnLineId → reservable left

    for (const lot of candidates) {
      if (need <= 0) break;
      // Serialize against transfers: lock the lot's GRN line FOR UPDATE on first
      // visit (scan order is deterministic across concurrent allocators).
      if (!lockedLines.has(lot.grnLineId)) {
        await tx
          .select({ id: grnLineItemsTable.id })
          .from(grnLineItemsTable)
          .where(eq(grnLineItemsTable.id, lot.grnLineId))
          .for("update")
          .limit(1);
        lockedLines.set(lot.grnLineId, (await lineAvailable(tx, lot.grnLineId)) - (await lineHeld(tx, lot.grnLineId)));
      }
      const reservable = lockedLines.get(lot.grnLineId)!;
      if (reservable <= 0) continue; // line consumed by transfers / other reservations
      const take = Math.min(need, reservable);
      await tx.insert(inventoryReservationAllocationsTable).values({
        reservationId: reservation.id,
        lotId: lot.id,
        quantity: String(take),
        status: "active",
      });
      allocations.push({
        lotId: lot.id,
        grnLineId: lot.grnLineId,
        lotNumber: lot.lotNumber,
        warehouseId: lot.warehouseId,
        locationId: lot.locationId,
        binId: lot.binId,
        quantity: take,
      });
      lockedLines.set(lot.grnLineId, reservable - take);
      need -= take;
    }

    if (need > 0 && !body.allow_partial) {
      return { status: "insufficient" as const, unallocated: need };
    }

    const newAllocated = allocated + allocations.reduce((s, a) => s + a.quantity, 0);
    const [updated] = await tx
      .update(inventoryReservationsTable)
      .set({ allocatedQty: String(newAllocated), status: nextStatus(reserved, newAllocated) })
      .where(eq(inventoryReservationsTable.id, reservation.id))
      .returning();

    await tx.insert(outboxEventsTable).values({
      aggregateType: "inventory_reservation",
      aggregateId: reservation.id,
      eventType: "RESERVATION_ALLOCATED",
      payload: {
        reservation_number: reservation.reservationNumber,
        strategy: body.strategy,
        allocations: allocations.map((a) => ({
          lot_id: a.lotId,
          lot_number: a.lotNumber,
          grn_line_id: a.grnLineId,
          quantity: a.quantity,
        })),
        new_allocated_qty: newAllocated,
        actor_id: actorId,
      },
    });

    return { status: "ok" as const, reservation: updated, allocations };

});

if (outcome.status === "not_found") { res.status(404).json({ error: "Reservation not found" }); return; }
if (outcome.status === "expired") { res.status(409).json({ error: "Reservation has expired" }); return; }
if (outcome.status === "invalid_state") { res.status(409).json({ error: `Reservation cannot be allocated from status '${outcome.current}'` }); return; }
if (outcome.status === "nothing") { res.status(422).json({ error: "Nothing left to allocate" }); return; }
if (outcome.status === "bad_lot") { res.status(422).json({ error: "Lot does not match the reserved material" }); return; }
if (outcome.status === "insufficient") {
res.status(409).json({ error: "INSUFFICIENT_STOCK", details: { unallocated: outcome.unallocated } });
return;
}

void recordSecurityEvent({
eventType: "reservation.allocated",
actorId,
actorEmail: req.user?.email ?? null,
actorRole: req.user?.role ?? null,
...reqMeta(req),
statusCode: 200,
detail: `Reservation ${outcome.reservation.reservationNumber} allocated across ${outcome.allocations.length} lot(s) [${body.strategy}]`,
});

res.json({
...serializeReservation(outcome.reservation as Record<string, any>),
allocations: outcome.allocations.map((a) => ({
lot_id: a.lotId,
lot_number: a.lotNumber,
grn_line_id: a.grnLineId,
quantity: a.quantity,
warehouse_id: a.warehouseId,
location_id: a.locationId,
bin_id: a.binId,
status: "active",
})),
});
});
