const jars = new Map<string, string>();

async function call(role: string | null, method: string, path: string, body?: unknown, headers?: Record<string, string>) {
const h: Record<string, string> = { ...(headers ?? {}) };
const cookie = role ? jars.get(role) : undefined;
if (cookie) h["cookie"] = cookie;
if (body !== undefined) h["content-type"] = "application/json";
const res = await fetch(BASE + "/api" + path, { method, headers: h, body: body !== undefined ? JSON.stringify(body) : undefined, redirect: "manual" });
const sc = res.headers.get("set-cookie");
if (role && sc) jars.set(role, sc.split(";")[0]);
const text = await res.text();
let json: any = null; try { json = JSON.parse(text); } catch {}
return { status: res.status, json };
}
async function login(role: string) {
const r = await call(role, "POST", "/auth/login", CRED[role]);
if (r.status !== 200) throw new Error(`login ${role}: HTTP ${r.status}`);
}
let passed = 0, failed = 0;
function check(id: string, cond: boolean, detail = "") {
if (cond) { passed++; console.log(`${id}: PASS`); } else { failed++; console.log(`${id}: FAIL ${detail}`); }
}

let supplierId: string | null = null, poId = "";
async function ensureSupplier(): Promise {
if (supplierId) return supplierId;

supplierId = r.json?.id ?? r.json?.data?.id;

}
async function makeMaterial(suffix: string): Promise<{ id: string; code: string }> {
await ensureSupplier();
const cat = await call("director", "POST", "/masters/material-categories", { code: `CERT70-CAT-${suffix}`, name: `Cat ${suffix}` });
const catId = cat.json?.id ?? cat.json?.data?.id;
await call("director", "POST", "/inventory/material-workflow-assignments", { category_id: catId, workflow_id: (await call("director", "POST", "/masters/material-workflows", { code: `CERT70-WF-${suffix}`, name: `WF ${suffix}`, post_receipt_action: "DIRECT_TO_INVENTORY" })).json?.id });
const code = `CERT70-MAT-${suffix}`;
const mat = await call("director", "POST", "/masters/materials", { code, name: `Mat ${suffix}`, category_id: catId, uom: "KG" });
const id = mat.json?.id ?? mat.json?.data?.id;
if (!id) throw new Error("material fixture failed");
return { id, code };
}
async function receive(materialId: string, qty: number, date?: string): Promise {
const c = await call("director", "POST", "/inventory/grns", { supplier_id: await ensureSupplier(), received_date: date ?? new Date().toISOString().slice(0, 10), lines: [{ material_id: materialId, quantity_received: qty }] });
const grnId = c.json?.id;
await call("director", "POST", `/inventory/grns/${grnId}/post`);
const d = await call("director", "GET", `/inventory/grns/${grnId}`);
return d.json?.lines?.[0]?.id;
}
async function ensurePO(): Promise {
if (poId) return poId;
const list = await call("director", "GET", "/manufacturing/orders?pageSize=1");
poId = list.json?.items?.[0]?.id;
if (!poId) poId = (await call("director", "POST", "/manufacturing/orders", { factoryManager: "cert-suite" })).json?.id;
if (!poId) throw new Error("PO fixture failed");
return poId;
}
async function reserveAllocate(materialId: string, qty: number): Promise {
const r = await call("supervisor", "POST", "/inventory/reservations", { production_order_id: await ensurePO(), material_id: materialId, quantity: qty });
await call("supervisor", "POST", `/inventory/reservations/${r.json.id}/allocate`);
return r.json.id as string;
}
async function stockQty(code: string, state: string): Promise {
const s = await call("viewer", "GET", `/inventory/stock?search=${encodeURIComponent(code)}&stock_state=${state}&pageSize=200`);
return (s.json?.items ?? []).reduce((t: number, r: any) => t + Number(r.quantity ?? 0), 0);
}
async function activeAllocSum(resId: string): Promise {
const d = await call("viewer", "GET", `/inventory/reservations/${resId}`);
return (d.json?.allocations ?? []).filter((a: any) => a.status === "active").reduce((s: number, a: any) => s + Number(a.quantity), 0);
}

-- INV-RES-01…08 (Task #69 baseline — must STILL be 0 rows; note INV-RES-01's

-- holding set now coexists with wip; the Phase 4 generalization is INV-P4-01)

(queries identical to task69_final_certification_report.md §2)

-- INV-P4-01: per material, reserved(holding) ≤ available + wip

SELECT material_id, SUM(held) held, SUM(avail) avail, SUM(wip) wip FROM (

SELECT material_id, SUM(reserved_qty)::numeric held, 0 avail, 0 wip

FROM inventory_reservations

WHERE status IN ('active','partially_allocated','fully_allocated','partially_issued','fully_issued')

GROUP BY material_id

UNION ALL

SELECT material_id, 0, SUM(quantity)::numeric FILTER (WHERE stock_state='available'), 0

FROM inventory_transactions GROUP BY material_id

UNION ALL

SELECT material_id, 0, 0, SUM(quantity)::numeric FILTER (WHERE stock_state='wip')

FROM inventory_transactions GROUP BY material_id) t

GROUP BY material_id HAVING SUM(held) > SUM(avail) + SUM(wip);

-- INV-P4-02: wip rows consistent and non-negative

SELECT id FROM wip_inventory

WHERE remaining_qty::numeric != issued_qty::numeric - consumed_qty::numeric

- returned_qty::numeric - scrapped_qty::numeric

OR remaining_qty::numeric < 0;

-- INV-P4-03: per (PO, material): Σ remaining (active wip) = Σ ledger wip

SELECT w.production_order_id, w.material_id,

SUM(w.remaining_qty)::numeric AS wip_rows,

(SELECT COALESCE(SUM(it.quantity),0)::numeric FROM inventory_transactions it

WHERE it.stock_state='wip' AND it.production_order_id = w.production_order_id

AND it.material_id = w.material_id) AS ledger_wip

FROM wip_inventory w

WHERE w.status IN ('active','partially_consumed')

GROUP BY w.production_order_id, w.material_id

HAVING SUM(w.remaining_qty)::numeric <> (

SELECT COALESCE(SUM(it.quantity),0)::numeric FROM inventory_transactions it

WHERE it.stock_state='wip' AND it.production_order_id = w.production_order_id

AND it.material_id = w.material_id);

-- INV-P4-04: variance identity + legal status transitions

SELECT id FROM consumption_confirmations

WHERE variance_qty::numeric != actual_qty::numeric - planned_qty::numeric;

SELECT id, status FROM consumption_confirmations

WHERE status NOT IN ('draft','confirmed','adjusted','rejected');

-- INV-P4-05: reservation ↔ allocation evidence equality

SELECT r.id FROM inventory_reservations r

WHERE (SELECT COALESCE(SUM(a.quantity),0)::numeric FROM inventory_reservation_allocations a

WHERE a.reservation_id = r.id AND a.status='active')

!= r.allocated_qty::numeric - r.issued_qty::numeric

OR (SELECT COALESCE(SUM(a.quantity),0)::numeric FROM inventory_reservation_allocations a

WHERE a.reservation_id = r.id AND a.status='issued')

!= r.issued_qty::numeric;

-- Ledger hygiene: no negative per-state balances; no reservation-sourced ledger rows

SELECT material_id, stock_state, SUM(quantity)::numeric FROM inventory_transactions

GROUP BY material_id, stock_state HAVING SUM(quantity) < 0;

SELECT count() FROM inventory_transactions

WHERE source_document_type IN ('reservation','reservation_allocation');  -- must be 0
