const jars = new Map<string, string>();

async function call(
role: string | null,
method: string,
path: string,
body?: unknown,
): Promise<{ status: number; json: any }> {
const headers: Record<string, string> = {};
const cookie = role ? jars.get(role) : undefined;
if (cookie) headers["cookie"] = cookie;
if (body !== undefined) headers["content-type"] = "application/json";
const res = await fetch(BASE + "/api" + path, {
method,
headers,
body: body !== undefined ? JSON.stringify(body) : undefined,
redirect: "manual",
});
const setCookie = res.headers.get("set-cookie");
if (role && setCookie) jars.set(role, setCookie.split(";")[0]);
const text = await res.text();
let json: any = null;
try { json = JSON.parse(text); } catch { / non-JSON / }
return { status: res.status, json };
}

async function makeMaterial(suffix: string, action: "DIRECT_TO_INVENTORY" | "INCOMING_INSPECTION"): Promise<{ id: string; code: string }> {
await ensureSupplier();
const cat = await call("director", "POST", "/masters/material-categories", { code: `CERT-CAT-${suffix}`, name: `Cert Cat ${suffix}` });
const catId = cat.json?.id ?? cat.json?.data?.id;
const wf = await call("director", "POST", "/masters/material-workflows", { code: `CERT-WF-${suffix}`, name: `Cert WF ${suffix}`, post_receipt_action: action }); // ADJUST field name if camelCase
const wfId = wf.json?.id ?? wf.json?.data?.id;
await call("director", "POST", "/inventory/material-workflow-assignments", { category_id: catId, workflow_id: wfId });
const code = `CERT-MAT-${suffix}`;
const mat = await call("director", "POST", "/masters/materials", { code, name: `Cert Mat ${suffix}`, category_id: catId, uom: "KG" });
const id = mat.json?.id ?? mat.json?.data?.id;

}

async function receive(materialId: string, qty: number, receivedDate?: string): Promise<{ grnId: string; lineId: string }> {
const sup = await ensureSupplier();
const c = await call("director", "POST", "/inventory/grns", {
supplier_id: sup,
received_date: receivedDate ?? new Date().toISOString().slice(0, 10),
lines: [{ material_id: materialId, quantity_received: qty }],
});
const grnId = c.json?.id;
if (!grnId) throw new Error(`GRN create failed: HTTP ${c.status} ${JSON.stringify(c.json)}`);
await call("director", "POST", `/inventory/grns/${grnId}/post`);
const d = await call("director", "GET", `/inventory/grns/${grnId}`);
return { grnId, lineId: d.json?.lines?.[0]?.id };
}

async function availableOf(code: string): Promise {
const s = await call("viewer", "GET", `/inventory/stock?search=${encodeURIComponent(code)}&stock_state=available&pageSize=200`);

}

-- INV-RES-01  reserved does not exceed reservable stock

SELECT material_id, SUM(available) available, SUM(reserved) reserved FROM (

SELECT material_id, SUM(quantity)::numeric available, 0 reserved

FROM inventory_transactions WHERE stock_state='available' GROUP BY material_id

UNION ALL

SELECT material_id, 0, SUM(reserved_qty)::numeric FROM inventory_reservations

WHERE status IN ('active','partially_allocated','fully_allocated','partially_issued','fully_issued')

GROUP BY material_id) t

GROUP BY material_id HAVING SUM(reserved) > SUM(available);

-- INV-RES-02  reservation quantities non-negative (DB CHECKs already enforce;

--             query proves it)

SELECT id FROM inventory_reservations

WHERE reserved_qty::numeric < 0 OR allocated_qty::numeric < 0 OR issued_qty::numeric < 0;

-- INV-RES-03  allocated ≤ reserved (CHECK enforced; query proves it)

SELECT id FROM inventory_reservations WHERE allocated_qty::numeric > reserved_qty::numeric;

-- INV-RES-04  no double allocation: active allocations per GRN line ≤ line available

SELECT ra.lot_id, SUM(ra.quantity)::numeric allocated FROM inventory_reservation_allocations ra

JOIN inventory_lots l ON l.id = ra.lot_id WHERE ra.status='active'

GROUP BY ra.lot_id, l.grn_line_id

HAVING SUM(ra.quantity)::numeric > (

SELECT COALESCE(SUM(it.quantity),0)::numeric FROM inventory_transactions it

WHERE it.source_line_id = l.grn_line_id AND it.stock_state='available');

-- INV-RES-05  no allocation from quarantined/rejected/consumed/expired lots

SELECT ra.id FROM inventory_reservation_allocations ra

JOIN inventory_lots l ON l.id = ra.lot_id

WHERE ra.status='active' AND l.status <> 'active';

-- INV-RES-06  lot/location dimensions reconcile: active allocations reference

--             a lot of the same material as the reservation

SELECT ra.id FROM inventory_reservation_allocations ra

JOIN inventory_lots l ON l.id = ra.lot_id

JOIN inventory_reservations r ON r.id = ra.reservation_id

WHERE ra.status='active' AND l.material_id <> r.material_id;

-- INV-RES-07  release/cancel restore reservable: terminal reservations keep no

--             active allocation

SELECT r.id FROM inventory_reservations r

WHERE r.status IN ('released','cancelled','expired')

AND EXISTS (SELECT 1 FROM inventory_reservation_allocations ra

WHERE ra.reservation_id = r.id AND ra.status='active');

-- INV-RES-08  ledger balances non-negative; reservations never wrote ledger rows

SELECT material_id, stock_state, SUM(quantity)::numeric balance

FROM inventory_transactions GROUP BY material_id, stock_state HAVING SUM(quantity) < 0;
