STATE MACHINE (Phase 0 schema, preserved)

request: draft → approved → in_transit(†issue) → received(†) → reconciled

[cancelled: draft|approved only] [rejected: in_transit only, and

only when no line has received anything]

line:    pending → issued → partially_received → fully_received [rejected]

†ISSUE  (physical): per line −qty 'available' @source dims / +qty

'in_transit' (TRANSFER_OUT). Full-line issue in one action.

†RECEIVE(physical): per line −qty 'in_transit' / +qty 'available' @dest

dims (TRANSFER_IN). PARTIAL receiving supported per line.

REJECT  (physical): −qty 'in_transit' / +qty 'available' @SOURCE dims

(TRANSFER_REVERSAL) for all issued-but-unreceived qty.

In-transit ledger rows carry source_document_id = request id and

source_line_id = the lot's GRN line → per-line conservation keyed by

(request, lot/grn-line) — see INV-P5-04.

LOCKS: issue: request FOR UPDATE → lines FOR UPDATE (lineNumber order) → per

line lot FOR UPDATE + GRN line FOR UPDATE (SAME objects transfers/allocate/

issue hold → full serialization; single-line lockers cannot deadlock against

this fixed order). receive/reject: request FOR UPDATE → lines FOR UPDATE

(in-transit is request-scoped; no external writer locks it). Idempotency:

Idempotency-Key on issue + receive ('transfer-idem:' / 'transfer-rcv-idem:')

— TOCTOU-hardened pattern (in-tx advisory lock + under-lock select +

dual-shape 23505 fallback); NO pre-tx lookup.

if (r.status === "in_transit") return { status: "already" as const };
if (r.status !== "approved") return { status: "invalid_state" as const, current: r.status };

    const lines = await tx.select().from(transferLinesTable)
      .where(eq(transferLinesTable.transferRequestId, r.id))
      .orderBy(asc(transferLinesTable.lineNumber)).for("update");

    for (const l of lines) {
      let sourceLineId: string | null = null;
      const conds = [
        eq(inventoryTransactionsTable.materialId, l.materialId),
        eq(inventoryTransactionsTable.stockState, "available"),
        eq(inventoryTransactionsTable.warehouseId, r.sourceWarehouseId),
      ];

      }

        .from(inventoryTransactionsTable).where(and(...conds));
      if (Number(l.requestedQty) > Number(bal.sum)) {
        return { status: "insufficient" as const, line: l.lineNumber, available: Number(bal.sum) };
      }
      await tx.insert(inventoryTransactionsTable).values({
        materialId: l.materialId, quantity: String(-Number(l.requestedQty)), uom: l.uom,
        stockState: "available", transactionType: "TRANSFER_OUT",
        sourceDocumentType: "transfer_request", sourceDocumentId: r.id, sourceLineId,

      await tx.insert(inventoryTransactionsTable).values({

        stockState: "in_transit", transactionType: "TRANSFER_OUT",
        sourceDocumentType: "transfer_request", sourceDocumentId: r.id, sourceLineId,
        lotId: l.lotId, warehouseId: r.sourceWarehouseId, locationId: l.sourceLocationId, binId: l.sourceBinId,
        actorId, actorName: req.user?.email ?? null,
      });
      await tx.update(transferLinesTable).set({
        issuedQty: l.requestedQty, status: "issued",
      }).where(eq(transferLinesTable.id, l.id));
    }

    const [u] = await tx.update(transferRequestsTable).set({
      status: "in_transit", issuedBy: actorId, issuedAt: new Date(),
      idempotencyKey: idemKey,

router.post("/:id/receive", requireWriteRole("supervisor", "director"), async (req: Request, res: Response): Promise => {
const parsed = ReceiveTransferBody.safeParse(req.body);
if (!parsed.success) { res.status(400).json({ error: parsed.error.message }); return; }

const idemKey = (req.header("idempotency-key") ?? "").slice(0, 100) || null;

if (idemKey) {
await tx.execute(sql`SELECT pg_advisory_xact_lock(hashtext(${`transfer-rcv-idem:{idemKey}`}))`);
const [replay] = await tx.select().from(transferRequestsTable)
.where(eq(transferRequestsTable.idempotencyKey, idemKey)).limit(1);
if (replay) return { status: "replay" as const, id: replay.id };
}
const [r] = await tx.select().from(transferRequestsTable)
.where(eq(transferRequestsTable.id, req.params.id as string)).for("update").limit(1);
if (!r) return { status: "not_found" as const };
if (r.status !== "in_transit") return { status: "invalid_state" as const, current: r.status };

    const lines = await tx.select().from(transferLinesTable)
      .where(and(
        eq(transferLinesTable.transferRequestId, r.id),
        inArray(transferLinesTable.id, parsed.data.lines.map((l) => l.line_id)),
      )).orderBy(asc(transferLinesTable.lineNumber)).for("update");

    for (const recv of parsed.data.lines) {
      const l = lines.find((x) => x.id === recv.line_id);
      if (!l) return { status: "bad_line" as const, line_id: recv.line_id };

      if (recv.quantity > outstanding) return { status: "over_receive" as const, line: l.lineNumber, outstanding };
      const newReceived = Number(l.receivedQty) + recv.quantity;
      let sourceLineId: string | null = null;
      if (l.lotId) {
        const [lot] = await tx.select().from(inventoryLotsTable)
          .where(eq(inventoryLotsTable.id, l.lotId)).limit(1);
        sourceLineId = lot?.grnLineId ?? null;
      }
      await tx.insert(inventoryTransactionsTable).values({
        materialId: l.materialId, quantity: String(-recv.quantity), uom: l.uom,
        stockState: "in_transit", transactionType: "TRANSFER_IN",
        sourceDocumentType: "transfer_request", sourceDocumentId: r.id, sourceLineId,
        lotId: l.lotId, warehouseId: r.sourceWarehouseId, locationId: l.sourceLocationId, binId: l.sourceBinId,
        actorId, actorName: req.user?.email ?? null,
      });
      await tx.insert(inventoryTransactionsTable).values({
        materialId: l.materialId, quantity: String(recv.quantity), uom: l.uom,
        stockState: "available", transactionType: "TRANSFER_IN",
        sourceDocumentType: "transfer_request", sourceDocumentId: r.id, sourceLineId,
        lotId: l.lotId, warehouseId: r.destinationWarehouseId,
        locationId: l.destinationLocationId, binId: l.destinationBinId,
        actorId, actorName: req.user?.email ?? null,
      });
      await tx.update(transferLinesTable).set({
        receivedQty: String(newReceived),
        status: newReceived >= Number(l.issuedQty) ? "fully_received" : "partially_received",
      }).where(eq(transferLinesTable.id, l.id));
    }

    const allLines = await tx.select().from(transferLinesTable)
      .where(eq(transferLinesTable.transferRequestId, r.id));
    const allReceived = allLines.every((l) => Number(l.receivedQty) >= Number(l.issuedQty));
    const [u] = await tx.update(transferRequestsTable).set({
      status: allReceived ? "received" : "in_transit",
      receivedBy: actorId, receivedAt: new Date(),
      idempotencyKey: idemKey,
    }).where(eq(transferRequestsTable.id, r.id)).returning();
    await tx.insert(outboxEventsTable).values({
      aggregateType: "transfer_request", aggregateId: r.id, eventType: "TRANSFER_RECEIVED",
      payload: { transfer_request_number: r.transferRequestNumber, partial: !allReceived, actor_id: actorId },
    });
    return { status: "ok" as const, id: u.id };

if (outcome.status === "not_found") { res.status(404).json({ error: "Transfer request not found" }); return; }
if (outcome.status === "replay") { const f = await loadRequest(outcome.id as string); res.status(200).json(serializeRequest(f!.r as Record<string, any>, f!.lines as Array<Record<string, any>>)); return; }

-- per (request, lot): issued = received + in-transit balance

SELECT l.transfer_request_id, l.lot_id,

MAX(l.issued_qty)::numeric AS issued, MAX(l.received_qty)::numeric AS received,

(SELECT COALESCE(SUM(it.quantity),0)::numeric FROM inventory_transactions it

WHERE it.source_document_type='transfer_request' AND it.source_document_id=l.transfer_request_id

AND it.stock_state='in_transit' AND (l.lot_id IS NULL OR it.lot_id = l.lot_id)) AS in_transit

FROM transfer_lines l WHERE l.issued_qty::numeric > 0

GROUP BY l.transfer_request_id, l.lot_id

HAVING MAX(l.issued_qty)::numeric <> MAX(l.received_qty)::numeric +

(SELECT COALESCE(SUM(it.quantity),0)::numeric FROM inventory_transactions it

WHERE it.source_document_type='transfer_request' AND it.source_document_id=l.transfer_request_id

AND it.stock_state='in_transit' AND (l.lot_id IS NULL OR it.lot_id = l.lot_id));
