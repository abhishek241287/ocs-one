const jars = new Map<string, string>();
async function call(role: string | null, method: string, path: string, body?: unknown, headers?: Record<string, string>) {
const h: Record<string, string> = { ...(headers ?? {}) };
const cookie = role ? jars.get(role) : undefined;
if (cookie) h["cookie"] = cookie;
if (body !== undefined) h["content-type"] = "application/json";
const res = await fetch(BASE + "/api" + path, { method, headers: h, body: body !== undefined ? JSON.stringify(body) : undefined, redirect: "manual" });
const sc = res.headers.get("set-cookie");
if (role && sc) jars.set(role, sc.split(";")[0]);
const text = await res.text();
let json: any = null; try { json = JSON.parse(text); } catch {}
return { status: res.status, json };
}
async function login(role: string) {
const r = await call(role, "POST", "/auth/login", CRED[role]);
if (r.status !== 200) throw new Error(`login ${role}: HTTP ${r.status}`);
}
let passed = 0, failed = 0;
function check(id: string, cond: boolean, detail = "") {
if (cond) { passed++; console.log(`${id}: PASS`); } else { failed++; console.log(`${id}: FAIL ${detail}`); }
}

// ─── fixtures ────────────────────────────────────────────────────────────────
let supplierId: string | null = null, poId = "", whA = "", whB = "";
async function ensureSupplier(): Promise {
if (supplierId) return supplierId;
const r = await call("director", "POST", "/masters/suppliers", { code: "CERT7K-SUP", name: "Cert7K Supplier" });
supplierId = r.json?.id ?? r.json?.data?.id;
if (!supplierId) throw new Error("supplier fixture failed");
return supplierId;
}
async function makeMaterial(suffix: string): Promise<{ id: string; code: string }> {
await ensureSupplier();
const cat = await call("director", "POST", "/masters/material-categories", { code: `CERT7K-CAT-${suffix}`, name: `Cat ${suffix}` });
const catId = cat.json?.id ?? cat.json?.data?.id;
const wf = await call("director", "POST", "/masters/material-workflows", { code: `CERT7K-WF-${suffix}`, name: `WF ${suffix}`, post_receipt_action: "DIRECT_TO_INVENTORY" });
await call("director", "POST", "/inventory/material-workflow-assignments", { category_id: catId, workflow_id: wf.json?.id });
const code = `CERT7K-MAT-${suffix}`;
const mat = await call("director", "POST", "/masters/materials", { code, name: `Mat ${suffix}`, category_id: catId, uom: "KG" });
const id = mat.json?.id ?? mat.json?.data?.id;
if (!id) throw new Error("material fixture failed");
return { id, code };
}
async function ensureWarehouses(): Promise {
if (whA) return;
const a = await call("director", "POST", "/inventory/warehouses", { code: "CERT7K-WHA", name: "Gate A", type: "main_store" });
const b = await call("director", "POST", "/inventory/warehouses", { code: "CERT7K-WHB", name: "Gate B", type: "main_store" });
whA = a.json?.id; whB = b.json?.id;
}
async function receive(materialId: string, qty: number): Promise {
const c = await call("director", "POST", "/inventory/grns", { supplier_id: await ensureSupplier(), received_date: new Date().toISOString().slice(0, 10), lines: [{ material_id: materialId, quantity_received: qty }] });
await call("director", "POST", `/inventory/grns/${c.json.id}/post`);
const d = await call("director", "GET", `/inventory/grns/${c.json.id}`);
return d.json?.lines?.[0]?.lot?.id ?? d.json?.lines?.[0]?.lot_id;
}
async function ensurePO(): Promise {
if (poId) return poId;
const list = await call("director", "GET", "/manufacturing/orders?pageSize=1");
poId = list.json?.items?.[0]?.id;
if (!poId) poId = (await call("director", "POST", "/manufacturing/orders", { factoryManager: "cert-7k" })).json?.id;
if (!poId) throw new Error("PO fixture failed");
return poId;
}
async function reserveIssue(materialId: string, qty: number): Promise<{ rid: string; noteId: string }> {
const r = await call("supervisor", "POST", "/inventory/reservations", { production_order_id: await ensurePO(), material_id: materialId, quantity: qty });
await call("supervisor", "POST", `/inventory/reservations/${r.json.id}/allocate`);
const i = await call("supervisor", "POST", `/inventory/reservations/${r.json.id}/issue`, {});
return { rid: r.json.id as string, noteId: i.json.id as string };
}
async function wipRow(materialId: string): Promise {
const w = await call("viewer", "GET", `/inventory/wip-inventory?material_id=${materialId}`);
return w.json?.items?.[0];
}
async function approvePost(path: string, body: unknown = {}): Promise {
await call("director", "POST", `${path}/approve`, {});
return call("supervisor", "POST", `${path}/post`, body);
}
async function stockQty(code: string, state: string): Promise {
const s = await call("viewer", "GET", `/inventory/stock?search=${encodeURIComponent(code)}&stock_state=${state}&pageSize=200`);
return (s.json?.items ?? []).reduce((t: number, r: any) => t + Number(r.quantity ?? 0), 0);
}
